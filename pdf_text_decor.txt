MVP: generate ONLY a simple flat brick wall in PixelLab, draw the floor in code (SimpleFloor, 
swappable later), and make furniture purchasable + draggable decor. You can build the 
WHOLE gameplay now with placeholder decor boxes and swap real art in later. Mixed 
currency: cheap = scrap, premium = premium gems. Placement is stored locally in Isar; 
ownership and spending go through the Firestore economy.
Decisions
 Placement: free drag-and-drop (Decorate mode)
 Scope: Living Room only (MVP)
 Currency: mixed - scrap (cheap) + premium gems (premium)
 Background: wall (PixelLab) + floor in code SimpleFloor (swap color/tile later, one block)

1) Art direction + sizes
 Wall: a simple flat brick wall only. View = Side (Sidescroller), Direction = None, Remove bg 
OFF.
 Wall in-game 400x350 (50/50 layout). Ideal canvas ~200x175 (ratio 8:7). 180x216 also 
works - minor brick stretch is barely visible.
 Floor: NOT generated; drawn in code via SimpleFloor. Changing color/tile later is one 
block.
 Decor sprites: one PNG per item, transparent, Remove bg ON. No need to make them yet - 
start with placeholder boxes (section 4), add art later.

class DecorDef {
  final String id;
  final String name;
  final String asset;            // example: decor/couch.png
  final DecorCategory category;
  final CurrencyType currency;   // scrap or premiumGems
  final int price;
  final Vector2 defaultSize;
  final Vector2 defaultPos;
  final int layer;               // render priority inside the room
  const DecorDef({
    required this.id,
    required this.name,
    required this.asset,
    required this.category,
    required this.currency,
    required this.price,
    required this.defaultSize,
    required this.defaultPos,
    this.layer = 0,
  });
}

4) DecorComponent (placeholder-ready)
class DecorComponent extends SpriteComponent
    with DragCallbacks, HasGameReference<RoboPetGame> {
  DecorComponent({required this.def, required Vector2 position, this.onMoved})
      : super(position: position, anchor: Anchor.bottomCenter, priority: def.layer);
  final DecorDef def;
  final void Function(DecorComponent)? onMoved;
  bool editable = false;
  @override
  Future<void> onLoad() async {
    size = def.defaultSize;
    try {
      sprite = await Sprite.load(def.asset);
    } catch (_) {
      // PLACEHOLDER: colored box + name until art exists.
      // When the PNG is added, the real sprite loads automatically; no code change.
      add(RectangleComponent(
        size: size,
        paint: Paint()..color = const Color(0xFF394067),
      ));
      add(TextComponent(text: def.name, anchor: Anchor.center, position: size / 2));
    }
  }
  // Only capture drags/taps in Decorate mode
  @override
  bool containsLocalPoint(Vector2 point) =>
      editable && super.containsLocalPoint(point);
  @override
  void onDragUpdate(DragUpdateEvent event) {
    if (!editable) return;
    position += event.localDelta;
    position.x = position.x.clamp(-180.0, 180.0);   // room width
    position.y = position.y.clamp(-300.0, 345.0);   // wall top to floor bottom
  }
  @override
  void onDragEnd(DragEndEvent event) {
    if (editable) onMoved?.call(this);
  }
}

// Simple, non-distracting floor. Change its color later, or swap to
// SpriteComponent('rooms/living_floor.png') for a tiled texture.
class SimpleFloor extends PositionComponent {
  SimpleFloor({required Vector2 size, required Vector2 position})
      : super(size: size, position: position, priority: -9);
  final _floor = Paint()..color = const Color(0xFF161A2E);
  final _shade = Paint()..color = const Color(0x22000000);
  final _edge = Paint()
    ..color = const Color(0xAA22E1FF)
    ..strokeWidth = 2;
  @override
  void render(Canvas canvas) {
    final w = size.x, h = size.y;
    canvas.drawRect(Rect.fromLTWH(0, 0, w, h), _floor);
    canvas.drawRect(Rect.fromLTWH(0, h * 0.5, w, h * 0.5), _shade);
    canvas.drawLine(const Offset(0, 0), Offset(w, 0), _edge);
  }
}
class LivingRoomWorld extends World with HasGameReference<RoboPetGame> {
  final List<DecorComponent> _decor = [];
  late final RobotDog _dog;
  @override
  Future<void> onLoad() async {
    // WALL: upper half (400x350)
    add(SpriteComponent(
      sprite: await Sprite.load('rooms/living_wall.png'),
      size: Vector2(400, 350),
      position: Vector2(-200, -350),
      priority: -10,
    ));
    // FLOOR: lower half, drawn in code (50/50)
    add(SimpleFloor(size: Vector2(400, 350), position: Vector2(-200, 0)));
    // saved decor
    final repo = game.ref.read(decorRepositoryProvider);
    for (final p in await repo.loadPlaced('living')) {
      final def = kLivingRoomDecor.firstWhere((d) => d.id == p.itemId);
      final c = DecorComponent(
        def: def,
        position: Vector2(p.x, p.y),
        onMoved: (comp) => repo.savePlacement('living', comp),
      );
      if (p.flipped) c.flipHorizontallyAroundCenter();
      _decor.add(c);
      add(c);
    }
    _dog = RobotDog(position: Vector2(0, 280));   // on the floor
    add(_dog);
  }
  void addDecor(DecorComponent c) {
    _decor.add(c);
    add(c);
    game.ref.read(decorRepositoryProvider).savePlacement('living', c);
  }
  void setEditMode(bool on) {
    for (final c in _decor) c.editable = on;
    if (on) { _dog.stop(); } else { _dog.startWander(); }
  }
}

6) Shop + purchase (economy-backed)
Future<bool> buyDecor(WidgetRef ref, DecorDef def) async {
  final economy = ref.read(economyServiceProvider);
  final ok = await economy.spend(
    currency: def.currency,   // scrap or premiumGems
    amount: def.price,
    reason: 'decor:${def.id}',
  );                           // bumps economyVersion, writes Firestore
  if (!ok) return false;       // not enough currency
  await economy.grantDecor(def.id);   // ownership in Firestore
  return true;
}
