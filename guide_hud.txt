2) Decay model
// Per-minute decay rates (tune freely)
const double kHungerPerMin = 1.2;
const double kHappyPerMin  = 0.8;
const double kEnergyPerMin = 1.0;
// Pure function: returns a new state decayed to now.
CareState decayed(CareState s, DateTime now) {
  final mins = now.difference(s.lastTick).inSeconds / 60.0;
  if (mins <= 0) return s;
  final drain = s.asleep ? 0.3 : 1.0; // drains slower while asleep
  return s.copyWith(
    hunger:    (s.hunger    - kHungerPerMin * mins * drain).round().clamp(0, 100),
    happiness: (s.happiness - kHappyPerMin  * mins * drain).round().clamp(0, 100),
    energy: s.asleep
        ? (s.energy + 2.0 * mins).round().clamp(0, 100)   // recharges while asleep
        : (s.energy - kEnergyPerMin * mins).round().clamp(0, 100),
    lastTick: now,
  );
}
3) CareNotifier (tick + offline catch-up)
class CareNotifier extends StateNotifier<CareState> {
  CareNotifier(this._isar) : super(_loadOrDefault(_isar)) {
    catchUp();                 // apply offline decay since last session
    _timer = Timer.periodic(const Duration(seconds: 30), (_) => _tick());
  }
  final Isar _isar;
  Timer? _timer;
  void _tick()   { state = decayed(state, DateTime.now()); _persist(); }
  void catchUp() { state = decayed(state, DateTime.now()); _persist(); }
  void feed()     => _bump(hunger: 30);
  void pet()      => _bump(happiness: 20);
  void recharge() => _bump(energy: 40);
  void _bump({int hunger = 0, int happiness = 0, int energy = 0}) {
    final s = decayed(state, DateTime.now()); // decay first, then add
    state = s.copyWith(
      hunger:    (s.hunger + hunger).clamp(0, 100),
      happiness: (s.happiness + happiness).clamp(0, 100),
      energy:    (s.energy + energy).clamp(0, 100),
      lastTick:  DateTime.now(),
    );
    _persist();
  }
  void _persist() { /* write state to Isar */ }
  @override
  void dispose() { _timer?.cancel(); super.dispose(); }
}
4) CareHud widget (stat bars)
class CareHud extends ConsumerWidget {
  const CareHud({super.key});
  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final s = ref.watch(careProvider);
    final warn = _firstCritical(s);
    return Positioned(
      top: 12, left: 12, right: 12,
      child: Column(children: [
        Row(children: [
          _StatBar(label: 'Hunger', value: s.hunger),
          const SizedBox(width: 8),
          _StatBar(label: 'Mood',   value: s.happiness),
          const SizedBox(width: 8),
          _StatBar(label: 'Energy', value: s.energy),
        ]),
        if (warn != null)
          Padding(
            padding: const EdgeInsets.only(top: 6),
            child: Text(warn, style: const TextStyle(
                color: Color(0xFFFF3CAC), fontSize: 12,
                fontWeight: FontWeight.bold)),
          ),
      ]),
    );
  }
  String? _firstCritical(CareState s) {
    if (s.energy < 20)    return 'Low battery! Charge your pet.';
    if (s.hunger < 20)    return 'Your pet is hungry!';
    if (s.happiness < 20) return 'Your pet feels lonely.';
    return null;
  }
}
class _StatBar extends StatelessWidget {
  const _StatBar({required this.label, required this.value});
  final String label;
  final int value;
  Color get _color {
    if (value < 20) return const Color(0xFFFF3CAC); // critical
    if (value < 50) return const Color(0xFFFFB627); // warning
    return const Color(0xFF7CFF6B);                  // good
  }
  @override
  Widget build(BuildContext context) {
    return Expanded(
      child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
        Text('\$label  \$value', style: const TextStyle(
            color: Colors.white, fontSize: 11)),
        const SizedBox(height: 3),
        ClipRRect(
          borderRadius: BorderRadius.circular(6),
          child: LinearProgressIndicator(
            value: value / 100.0,
            minHeight: 7,
            backgroundColor: const Color(0x33FFFFFF),
            valueColor: AlwaysStoppedAnimation<Color>(_color),
          ),
        ),
      ]),
    );
  }
}
