Part A - Combat balance and progression
Why Level 1 is currently unbeatable
With the current stats (player ATK 15 / DEF 10 / regen 2, enemy HP 120 / ATK 18 / DEF 12) the 
enemy deals 18 - 10 x 0.3 = 15 per tick, minus 2 regen = 13 net, so the player (HP 100) dies in 
about 8 ticks. The player deals 15 - 12 x 0.3 = 11 per tick and needs about 11 ticks to drop the 
enemy. Because both sides tick at the same speed, the player always dies first - Level 1 is 
mathematically unwinnable. The fix below makes Level 1 enemies weak and scales difficulty up 
per level.
Level-scaled enemy
class EnemyConfig {
  final int hp;
  final int atk;
  final int def;
  final int spd;
  const EnemyConfig({
    required this.hp,
    required this.atk,
    required this.def,
    required this.spd,
  });
}
// Level 1 is intentionally easy so new players can win and learn the loop.
EnemyConfig enemyForLevel(int level) {
  final l = (level - 1).clamp(0, 999);
  return EnemyConfig(
    hp: 55 + l * 25,
    atk: 9 + l * 3,
    def: 5 + l * 2,
    spd: 9 + l,
  );
}
Level 1 enemy becomes HP 55 / ATK 9 / DEF 5. The player now deals about 14 per tick (kills in ~4 
ticks) and only takes ~4 net per tick, so Level 1 is a comfortable win.
Start a fight from the current level
void startFight() {
  // state.level is the current unlocked level and starts at 1.
  final cfg = enemyForLevel(state.level);
  state = state.copyWith(
    enemyMaxHp: cfg.hp,
    enemyHp: cfg.hp,
    enemyAtk: cfg.atk,
    enemyDef: cfg.def,
    enemySpd: cfg.spd,
    playerHp: state.playerMaxHp,
    phase: CombatPhase.fighting,
  );
  _timer = Timer.periodic(const Duration(milliseconds: 1600), (_) => _tick());
}
Resolve each tick, then win or lose
void _tick() {
  // Player hits enemy.
  final toEnemy = (state.playerAtk - state.enemyDef * 0.3).round().clamp(1, 9999);
  final enemyHp = (state.enemyHp - toEnemy).clamp(0, state.enemyMaxHp);
  // Enemy hits player, then the player regenerates a little.
  final toPlayer = (state.enemyAtk - state.playerDef * 0.3).round().clamp(1, 9999);
  final playerHp =
      (state.playerHp - toPlayer + state.playerRegen).clamp(0, state.playerMaxHp);
  state = state.copyWith(enemyHp: enemyHp, playerHp: playerHp);
  if (enemyHp <= 0) {
    _onVictory();
  } else if (playerHp <= 0) {
    _onDefeat();
  }
}
void _onVictory() {
  _timer?.cancel();
  final reward = 10 + state.level * 5;
  ref.read(economyProvider.notifier).addScrap(reward);
  state = state.copyWith(
    level: state.level + 1, // unlock next level
    phase: CombatPhase.victory,
    lastReward: reward,
  );
}
void _onDefeat() {
  _timer?.cancel();
  state = state.copyWith(phase: CombatPhase.defeat);
}
void retry() {
  // Same level, full HP, fight again.
  startFight();
}
