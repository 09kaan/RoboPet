5.1 RobotDog - tap + reaction
class RobotDog extends SpriteAnimationGroupComponent<DogAnim>
    with HasGameReference<RoboPetGame>, TapCallbacks {
  RobotDog({required Vector2 position})
      : super(position: position, size: Vector2.all(160),
              anchor: Anchor.bottomCenter, priority: 10);
  bool _asleep = false;
  // Play an animation for a fixed duration, then return to idle
  Future<void> reactFor(DogAnim anim, double seconds) async {
    if (_asleep) return;
    stopWander();
    current = anim;
    await Future<void>.delayed(
        Duration(milliseconds: (seconds * 1000).round()));
    current = DogAnim.idle;
    startWander();
  }
  Future<void> reactEat()   => reactFor(DogAnim.eat, 2.5);
  Future<void> reactHappy() => reactFor(DogAnim.happy, 2.0);
  @override
  void onTapUp(TapUpEvent event) {
    if (game.decorateMode) return;   // ignore taps while decorating
    game.openDogActions();
  }
}
5.2 Action menu overlay
class DogActionBar extends ConsumerWidget {
  const DogActionBar({super.key, required this.game});
  final RoboPetGame game;
  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final care = ref.read(careProvider.notifier);
    Widget btn(String icon, String label, VoidCallback onTap) =>
        GestureDetector(
          onTap: () { onTap(); game.closeDogActions(); },
          child: Column(mainAxisSize: MainAxisSize.min, children: [
            Text(icon, style: const TextStyle(fontSize: 28)),
            Text(label, style: const TextStyle(
                color: Colors.white, fontSize: 12)),
          ]),
        );
    return Align(
      alignment: Alignment.bottomCenter,
      child: Container(
        margin: const EdgeInsets.only(bottom: 90),
        padding: const EdgeInsets.symmetric(horizontal: 24, vertical: 14),
        decoration: BoxDecoration(
          color: const Color(0xEE0B0E1A),
          borderRadius: BorderRadius.circular(20),
          border: Border.all(color: const Color(0xFF22E1FF)),
        ),
        child: Row(mainAxisSize: MainAxisSize.min, children: [
          btn('food', 'Feed',   () { care.feed();     game.dog.reactEat(); }),
          const SizedBox(width: 28),
          btn('pet',  'Pet',    () { care.pet();      game.dog.reactHappy(); }),
          const SizedBox(width: 28),
          btn('bolt', 'Charge', () { care.recharge(); game.dog.reactHappy(); }),
        ]),
      ),
    );
  }
}

5.3 Care integration (adapt to your CareNotifier)
// CareNotifier (Phase 1) - adapt field names to your real CareState
void feed()     => _bump(hunger: 30);
void pet()      => _bump(happiness: 20);
void recharge() => _bump(energy: 40);
void _bump({int hunger = 0, int happiness = 0, int energy = 0}) {
  state = state.copyWith(
    hunger:    (state.hunger + hunger).clamp(0, 100),
    happiness: (state.happiness + happiness).clamp(0, 100),
    energy:    (state.energy + energy).clamp(0, 100),
    lastInteraction: DateTime.now(),
  );
  _persist(); // write to Isar
}
5.4 Animation sync (battery / sleep)
// call when care changes:
// ref.listen(careProvider, (_, s) => dog.applyCareState(
//     energy: s.energy, asleep: s.asleep));
void applyCareState({required int energy, required bool asleep}) {
  _asleep = asleep;
  if (asleep) { current = DogAnim.sleep; stopWander(); return; }
  if (energy < 20) {
    current = DogAnim.lowPower;
  } else if (current == DogAnim.sleep || current == DogAnim.lowPower) {
    current = DogAnim.idle; startWander();
  }
}
